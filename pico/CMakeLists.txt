cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w CACHE STRING "Board")
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(snos_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(SKIP_PICO_BTSTACK 1) # No Bluetooth targets.
pico_sdk_init()
add_executable(snos_pico main.c port.c link.c ctrl.c kbd.c relay.c config.c log.c)
pico_generate_pio_header(snos_pico ${CMAKE_CURRENT_SOURCE_DIR}/snoslink.pio)
target_include_directories(snos_pico PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(snos_pico PRIVATE DATA1_OPEN_DRAIN=1)
target_compile_options(snos_pico PRIVATE -Wall -Wextra)
target_link_libraries(snos_pico pico_stdlib hardware_pio hardware_dma hardware_flash pico_cyw43_arch_lwip_threadsafe_background)
pico_enable_stdio_uart(snos_pico 1)
pico_enable_stdio_usb(snos_pico 1)
target_link_options(snos_pico PRIVATE "-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/config_flash.ld")
pico_add_extra_outputs(snos_pico)

add_executable(snos_wiretest wiretest.c port.c)
pico_generate_pio_header(snos_wiretest ${CMAKE_CURRENT_SOURCE_DIR}/snoslink.pio)
target_include_directories(snos_wiretest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(snos_wiretest PRIVATE DATA1_OPEN_DRAIN=1)
target_compile_options(snos_wiretest PRIVATE -Wall -Wextra)
target_link_libraries(snos_wiretest pico_stdlib hardware_pio hardware_dma)
pico_enable_stdio_uart(snos_wiretest 1)
pico_enable_stdio_usb(snos_wiretest 1)
pico_add_extra_outputs(snos_wiretest)

add_executable(snos_wiretest_pp wiretest.c port.c)
pico_generate_pio_header(snos_wiretest_pp ${CMAKE_CURRENT_SOURCE_DIR}/snoslink.pio)
target_include_directories(snos_wiretest_pp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
# Push-pull DATA1: only for a wiring with a 5 V buffer (74AHCT125) on GP5.
target_compile_options(snos_wiretest_pp PRIVATE -Wall -Wextra)
target_link_libraries(snos_wiretest_pp pico_stdlib hardware_pio hardware_dma)
pico_enable_stdio_uart(snos_wiretest_pp 1)
pico_enable_stdio_usb(snos_wiretest_pp 1)
pico_add_extra_outputs(snos_wiretest_pp)
